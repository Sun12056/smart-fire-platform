## 3. F5 实现（关键代码）

```js
// 无有效方向时的中性占位：虚线圆环，**只用 arc**（不产生任何箭头/直线段几何）
export function makeNeutralDirectionTexture(color = '#39FF88') {
  const size = 128
  const c = document.createElement('canvas')
  c.width = size
  c.height = size
  const ctx = c.getContext('2d')
  ctx.clearRect(0, 0, size, size)
  ctx.strokeStyle = color
  ctx.lineWidth = 10
  ctx.setLineDash([10, 10])
  ctx.beginPath()
  ctx.arc(size / 2, size / 2, 34, 0, Math.PI * 2)
  ctx.stroke()
  const tex = new THREE.CanvasTexture(c)
  tex.colorSpace = THREE.SRGBColorSpace
  tex.needsUpdate = true
  return tex
}
```

- 与 `makeArrowTexture` 同为 128×128，Sprite 缩放（0.8）无需调整；
- 半径 34 / 线宽 10 / 虚线 `[10,10]` 对应设计中「虚线圆环、无指向性几何」。

## 4. F4 实现（关键代码）

```js
  /** 方向显示判定：裁决表索引（S2 导出）+ 设备历史值 → 统一 resolver；索引缺失按 null ⇒ fail-closed */
  _displayOf(dev, store) {
    const index = store && store.directionVerdictIndex
    const verdict = index ? (index.get(String(dev.id)) || null) : null
    return resolveDirectionDisplay(verdict, dev.direction)
  }

  /** 渲染方向 → 贴图；空串（无有效方向）⇒ 中性占位圆环 */
  _textureOf(dir) {
    if (dir === 'left') return this.arrowTexL
    if (dir === 'right') return this.arrowTexR
    return this.neutralTex
  }

  _bindEvac(dev, store) {
    const disp = this._displayOf(dev, store)
    const dir = disp.direction || ''
    let rec = this.evacMap.get(dev.id)
    if (rec) {
      if (rec.dir !== dir) {
        rec.sprite.material.map = this._textureOf(dir)
        rec.sprite.material.needsUpdate = true
        rec.dir = dir
      }
      rec.device = dev
      rec.displayState = disp.state
      rec.verdict = disp
      return
    }
    const pos = this._resolveDevicePos(dev, 0.6)
    if (!pos) return
    const spr = new THREE.Sprite(new THREE.SpriteMaterial({
      map: this._textureOf(dir), color: 0xffffff, transparent: true, depthWrite: false,
    }))
    spr.scale.set(0.8, 0.8, 1)
    spr.position.set(pos.x, pos.top + 0.75, pos.z)
    this.evacGroup.add(spr)
    rec = { id: String(dev.id), sprite: spr, dir, device: dev, displayState: disp.state, verdict: disp }
    this.evacMap.set(dev.id, rec)
  }
```

## 6. `rec.runtime.direction` 权威值保持验证

| 断言 | 结论 |
|---|---|
| `_syncRuntime` 仍为 `const rt = deviceRuntimeState(dev)` + `Object.assign(rec.runtime, rt)`（T3 D16） | ✅ 未改 |

| 渲染与运行态解耦（T3 D23） | **实测修正了本轮一个错误假设**：`deviceRuntimeState.direction` = `deviceDirection(dev)`，即 **B4-03 折叠值**（`'up'` → `'right'`），并非原始字段。故 D23 改为断言「运行态沿用折叠（`up→right`）、渲染侧对同一值 fail-closed（`null`）」，二者互不覆盖 |

> 这条必须记录清楚：`rec.runtime.direction` **不是**原始 `dev.direction` 的镜像，而是既有 B4-03 投影。任何"用 runtime.direction 判断合法性"的后续需求都不成立；合法性判定只属于 resolver（`dev.direction` 原值 + 裁决表）。
